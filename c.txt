import org.arl.fjage.*
import org.arl.fjage.param.Parameter
import org.arl.unet.*
import org.arl.unet.link.LinkStatusNtf
import org.arl.unet.net.*
import org.arl.unet.nodeinfo.NodeInfoParam
import java.util.Random

/**
 * Live UnetStack implementation of EAFTR. One instance runs at every node,
 * learns a local Q-learning policy and adds the selected next hop to Router.
 * Direct neighbours exchange lightweight request/reply energy datagrams; no
 * agent reads another agent's fields or Q-table.
 */
class EnergyAwareRoutingAgent extends UnetAgent {
  enum Params implements Parameter { destination, energy, failedLinks, failedNodes, episodes, neighborInfo, qtable }
  static final String title = 'EAFTR Q-learning routing'
  static final String description = 'Energy-aware, fault-tolerant routing policy'
  double alpha=0.15d, gamma=0.90d, epsilonStart=0.30d, epsilonEnd=0.03d
  // Keep the live policy's feasibility and reward model identical to the
  // offline experiment: 128 bytes at 10 kbps, 10 W TX and 2 W RX.
  static final double packetBytes=128d, dataRate=10000d, txPower=10d, rxPower=2d
  static final double routingCost=0.05d, energyReserve=0.10d
  static final int energyProtocol=Protocol.USER+1
  // 800 episodes keeps local policy initialization responsive while still
  // providing epsilon-greedy model-bootstrap convergence.
  int episodes=800, destination=8
  // Each physical node owns only this agent's energy model. It is not a
  // hardware measurement and is never a network-wide vector.
  private double initialEnergy=100d
  private double residualEnergy=100d
  Set<String> failedLinks=[] as Set<String>
  Set<Integer> failedNodes=[] as Set<Integer>

  final Map<Integer,List<Integer>> neighbors=[
    1:[2,3], 2:[1,4,5], 3:[1,5,6], 4:[2,7],
    5:[2,3,7], 6:[3,7], 7:[4,5,6,8], 8:[7]]
  final Map<Integer,List<Double>> position=[
    1:[0d,0d,-10d], 2:[100d,0d,-15d], 3:[0d,100d,-20d],
    4:[200d,0d,-20d], 5:[100d,100d,-25d], 6:[0d,200d,-15d],
    7:[250d,100d,-20d], 8:[350d,100d,-10d]]
  private AgentID router, link
  private int address=-1
  // Q[destination][nextHop]; the current node is implicit in this instance.
  private Map<Integer,Map<Integer,Double>> qTables=[:]
  // Actions are bootstrapped once when they first become feasible. This is
  // separate from table creation because neighbour replies arrive later.
  private Map<Integer,Set<Integer>> bootstrappedActions=[:]
  // Snapshots received in EAFTR request/reply datagrams. They are local,
  // expire after 90 s, and are never read from another agent.
  private Map<Integer,Map> neighborCache=[:]
  static final long neighborInfoMaxAgeMs=90000L
  // The wall-clock guard prevents control traffic from draining the model when
  // a simulator schedules wakeups faster than expected.
  static final long neighborQueryIntervalMs=60000L
  private long localAdvertisementSequence=0L
  private Map<Integer,Long> lastEnergyRequestAt=[:]
  private Map<Integer,Double> pendingControlTxCost=[:]
  private Map<Integer,Integer> installedNextHops=[:]
  private boolean refreshQueued=false

  @Override List<Parameter> getParameterList() { allOf(Params) }
  // Unet 7.0.0 does not declare this hook on UnetAgent, so this intentionally
  // has no @Override annotation. Newer shells still discover it as a property
  // metadata method.
  boolean isReadOnly(Parameter parameter) {
    parameter == Params.qtable || parameter == Params.neighborInfo
  }
  double getEnergy() { residualEnergy }
  String getNeighborInfo() {
    neighborCache.collect { int id, Map n -> "${id}:${n.energy}:${n.sequence}:${n.time}" }.join(',')
  }
  String getQtable() {
    if (address < 1 || qTables.isEmpty()) return 'No local Q-values yet; wait for startup and fresh neighbour information.'
    StringBuilder out=new StringBuilder("EAFTR local Q-table at node ${address} (Q[destination][nextHop])\n")
    qTables.keySet().sort().each { int target ->
      out.append("\nDestination ${target}\n")
      out.append('  ').append(qTables[target].keySet().sort().collect { int hop ->
        "${hop}:${String.format('%.2f', qTables[target][hop])}"
      }.join(', ')).append('\n')
    }
    out.toString()
  }
  @Override void startup() {
    router=agentForService(Services.ROUTING); link=agentForService(Services.LINK)
    if (link != null) subscribe topic(link)
    // UnetStack 7 publishes received datagrams on this global topic.
    subscribe topic(Topics.DATAGRAM)
    add new WakerBehavior(1000, {
      def info=agentForService(Services.NODE_INFO)
      address=get(info,NodeInfoParam.address) as int
      refreshPolicy()
      requestNeighborEnergy()
      scheduleNeighborEnergyRequests()
    })
  }
  @Override void processMessage(Message msg) {
    if (address<1) return
    if (msg instanceof DatagramNtf) {
      // A DatagramNtf confirms successful local delivery, so RX is charged
      // exactly once here, including for EAFTR control datagrams.
      double receiveCost=msg.protocol==energyProtocol ? rxEnergy(msg.data?.length ?: 0d) : rxEnergy()
      debitLocalEnergy(receiveCost, 'verified datagram reception')
      if (msg.protocol == energyProtocol) handleEnergyDatagram(msg)
      return
    }
    if (msg.sender != link) return
    // One local ECLink notification represents one actual acoustic send
    // attempt. Retries are separate attempts and are charged separately.
    if (msg instanceof DatagramTransmissionNtf) {
      Double controlCost=pendingControlTxCost.remove(msg.to)
      debitLocalEnergy(controlCost ?: routingCost+txEnergy(),
        controlCost == null ? 'data transmission decision and TX' : 'energy-control TX')
      return
    }
    if (!(msg instanceof LinkStatusNtf)) return
    if (!neighbors[address]?.contains(msg.to)) return
    String edge=key(address,msg.to)
    if (msg.up) failedLinks.remove(edge); else failedLinks.add(edge)
    queueRefresh()
  }
  /** Configure this node only; a list is rejected to prevent shared energy. */
  void setEnergy(Object value) {
    if (!(value instanceof Number) || (value as double)<0d)
      throw new IllegalArgumentException('energy must be one non-negative joule value for this node')
    initialEnergy=value as double; residualEnergy=initialEnergy; queueRefresh()
  }
  void setFailedLinks(Collection values) {
    this.@failedLinks=(values ?: []).collect { it.toString() } as Set<String>; queueRefresh()
  }
  void setFailedNodes(Collection values) {
    this.@failedNodes=(values ?: []).collect { it as int } as Set<Integer>; queueRefresh()
  }
  void setDestination(int value) {
    if (!neighbors.containsKey(value)) throw new IllegalArgumentException('unknown destination')
    this.@destination=value; queueRefresh()
  }
  void setEpisodes(int value) {
    if (value<1) throw new IllegalArgumentException('episodes must be positive')
    this.@episodes=value; queueRefresh()
  }
  private void queueRefresh() {
    if (address<1 || refreshQueued) return
    refreshQueued=true
    add new WakerBehavior(250, { refreshQueued=false; refreshPolicy() })
  }
  /** Optional hook for a verified reception path outside DatagramNtf. */
  void accountReception() { debitLocalEnergy(rxEnergy(), 'verified reception') }
  /** Stores a validated EAFTR reply locally; it never accesses a remote field. */
  void recordNeighborAdvertisement(int node, double joules, long sequence, long timestamp=System.currentTimeMillis()) {
    if (address<1 || !neighbors[address]?.contains(node) || joules<0d || timestamp<=0L) return
    Map previous=neighborCache[node]
    if (previous != null && sequence <= (previous.sequence as long)) return
    neighborCache[node]=[energy:joules, sequence:sequence, time:timestamp]
    log.info(String.format('EAFTR: node %d cached %.3f J from neighbour %d',address,joules,node))
    queueRefresh()
  }
  private void debitLocalEnergy(double amount, String event) {
    if (amount<=0d) return
    residualEnergy=Math.max(0d,residualEnergy-amount)
    log.info(String.format('EAFTR: node %d energy %.3f J after %s',address,residualEnergy,event))
    queueRefresh()
  }
  private void scheduleNeighborEnergyRequests() {
    add new WakerBehavior(neighborQueryIntervalMs, {
      if (address>0 && !failedNodes.contains(address)) requestNeighborEnergy()
      scheduleNeighborEnergyRequests()
    })
  }
  /** Requests each direct neighbour's own estimate over one-hop ECLink. */
  private void requestNeighborEnergy() {
    if (link==null || !canTransmitLocal()) return
    long requestId=++localAdvertisementSequence
    long now=System.currentTimeMillis()
    neighbors[address]?.each { int node ->
      Long previous=lastEnergyRequestAt[node]
      if (!failedNodes.contains(node) && !failedLinks.contains(key(address,node)) &&
          (previous==null || now-previous>=neighborQueryIntervalMs)) {
        lastEnergyRequestAt[node]=now
        log.info("EAFTR: node ${address} requesting energy from neighbour ${node}")
        sendEnergyControl(node,"EAFTR:ENERGY_REQ:${requestId}")
      }
    }
  }
  private void handleEnergyDatagram(DatagramNtf msg) {
    String payload
    try { payload=new String(msg.data as byte[], 'UTF-8') }
    catch (Exception ignored) { return }
    String[] part=payload.split(':')
    if (part.size()==3 && part[0]=='EAFTR' && part[1]=='ENERGY_REQ') {
      // The normal local TX notification charges this reply's decision + TX.
      if (neighbors[address]?.contains(msg.from) && canTransmitLocal()) {
        long sequence=++localAdvertisementSequence
        log.info("EAFTR: node ${address} replying to energy request from ${msg.from}")
        sendEnergyControl(msg.from,"EAFTR:ENERGY_RSP:${sequence}:${residualEnergy}:${System.currentTimeMillis()}")
      }
      return
    }
    if (part.size()==5 && part[0]=='EAFTR' && part[1]=='ENERGY_RSP') {
      try {
        // Reply layout is EAFTR:ENERGY_RSP:sequence:joules:timestamp.
        recordNeighborAdvertisement(msg.from as int,part[3] as double,part[2] as long,part[4] as long)
      } catch (Exception ignored) { log.warning('EAFTR: ignored malformed neighbour-energy response') }
    }
  }
  private void sendEnergyControl(int target, String payload) {
    byte[] data=payload.getBytes('UTF-8')
    pendingControlTxCost[target]=txEnergy(data.length as double)
    send new DatagramReq(recipient:link, to:target, protocol:energyProtocol,
      reliability:false, data:data)
  }

  /**
   * Installs a route to every other address in the known topology. This makes
   * the agent a real Router policy, not a N0-to-N7-only demonstration.
   */
  void refreshPolicy() {
    if (router==null || address<1) return
    // Remove routes that now point at a declared failed destination. If this
    // node itself is failed, it must not retain any EAFTR forwarding policy.
    installedNextHops.keySet().findAll { failedNodes.contains(it) || failedNodes.contains(address) }
      .toList().each { int target -> removeInstalledRoute(target) }
    if (failedNodes.contains(address)) return
    neighbors.keySet().findAll { it!=address && !failedNodes.contains(it) }
      .sort().each { int target -> installRoute(target) }
  }
  private void installRoute(int target) {
    Map<Integer,Double> values=localTable(target)
    // A refresh retains values; only newly initialized local actions are
    // model-bootstrapped below.
    trainUnseenActions(target,values)
    Integer next=bestLocalAction(target,values)
    if (next==null) {
      // A route learned before a failure/energy change must not survive as a
      // stale Router entry. RDP routes are untouched.
      removeInstalledRoute(target)
      log.warning("EAFTR: no feasible path from ${address} to ${target}")
      return
    }
    // Delete previous policy route for this destination but retain RDP fallback routes.
    removeInstalledRoute(target)
    def request=EditRouteReq.newRoute()
    request.to=target; request.nextHop=next; request.link=link; request.hops=1
    request.reliability=true; request.metric=(1000d+values[next]) as float
    request.auto=false; request.recipient=router; send(request)
    installedNextHops[target]=next
    log.info("EAFTR: route ${address} -> ${target} via ${next}")
  }
  private void removeInstalledRoute(int target) {
    Integer oldNext=installedNextHops.remove(target)
    if (oldNext == null) return
    def remove=EditRouteReq.deleteRoute()
    remove.to=target; remove.nextHop=oldNext; remove.link=link
    remove.recipient=router; send(remove)
  }

  private Map<Integer,Double> localTable(int target) {
    if (!qTables.containsKey(target)) qTables[target]=neighbors[address].collectEntries { int hop -> [(hop):0d] }
    qTables[target]
  }
  /**
   * The available live stack has no portable delivery/ACK feedback. Bootstrap
   * is therefore a terminal one-hop model transition (future max = 0), never
   * a fabricated read of the next node's Q-table. Values persist afterwards.
   */
  private void trainUnseenActions(int target, Map<Integer,Double> values) {
    List<Integer> choices=actions(target,[] as Set<Integer>)
    if (!choices) return
    Set<Integer> trained=bootstrappedActions[target]
    if (trained==null) { trained=[] as Set<Integer>; bootstrappedActions[target]=trained }
    choices.findAll { !trained.contains(it) }.each { int next ->
      // Model bootstrap for this newly feasible action only. The value remains
      // unchanged on ordinary refreshes and later comes from real feedback.
      episodes.times {
        values[next]+=alpha*(reward(address,next,target)-values[next])
      }
      trained.add(next)
    }
  }
  /** Hook for a deployment-specific ACK/delivery adapter with real feedback. */
  void recordForwardingFeedback(int target, int nextHop, double observedReward, double futureMax=0d, boolean terminal=true) {
    Map<Integer,Double> values=localTable(target)
    if (!values.containsKey(nextHop)) return
    double continuation=terminal ? 0d : Math.max(0d,futureMax)
    values[nextHop]+=alpha*(observedReward+gamma*continuation-values[nextHop])
    queueRefresh()
  }
  private Integer bestLocalAction(int target, Map<Integer,Double> values) {
    List<Integer> options=actions(target,[] as Set<Integer>)
    options ? maxQ(options,values) : null
  }
  private int maxQ(List<Integer> options, Map<Integer,Double> values) {
    double high=options.collect { values[it] ?: 0d }.max() as double
    options.find { Math.abs((values[it] ?: 0d)-high)<1e-9 }
  }
  // Only this node's energy and direct-neighbour advertisements are used.
  // Remote battery state is intentionally unknown to this local agent.
  private List<Integer> actions(int target, Set<Integer> visited) {
    if (failedNodes.contains(address) || !canTransmitLocal()) return []
    neighbors[address].findAll { int to ->
      !visited.contains(to) && !failedNodes.contains(to) && !failedLinks.contains(key(address,to)) &&
        neighborCanReceive(to,target) && pathExists(to,visited,target)
    }
  }
  private boolean pathExists(int start, Set<Integer> visited, int target) {
    def queue=new ArrayDeque<Integer>(); def seen=new LinkedHashSet<Integer>()
    queue.add(start); seen.add(start)
    while (!queue.isEmpty()) {
      int current=queue.removeFirst()
      if (current==target) return true
      neighbors[current].each { int next ->
        if (!seen.contains(next) && !visited.contains(next) && !failedNodes.contains(next) &&
            !failedLinks.contains(key(current,next))) { seen.add(next); queue.add(next) }
      }
    }
    false
  }
  private double reward(int from, int to, int target) {
    if (to==target) return 100d
    double battery=10d*freshNeighborEnergy(to)/Math.max(1d,initialEnergy)
    double quality=5d*Math.exp(-distance(from,to)/500d)
    double progress=(distance(from,target)-distance(to,target))/100d
    battery+quality+progress-2d*(txEnergy()+rxEnergy()+routingCost)-1d
  }
  private boolean canTransmitLocal() { residualEnergy>=routingCost+txEnergy()+energyReserve }
  private boolean neighborCanReceive(int node, int target) {
    freshNeighborEnergy(node)>=(node==target ? rxEnergy()+energyReserve : rxEnergy()+routingCost+txEnergy()+energyReserve)
  }
  private double freshNeighborEnergy(int node) {
    Map snapshot=neighborCache[node]
    if (snapshot==null || System.currentTimeMillis()-(snapshot.time as long)>neighborInfoMaxAgeMs) return -1d
    snapshot.energy as double
  }
  private static double packetDuration(double bytes=packetBytes) { bytes*8d/dataRate }
  private static double txEnergy(double bytes=packetBytes) { txPower*packetDuration(bytes) }
  private static double rxEnergy(double bytes=packetBytes) { rxPower*packetDuration(bytes) }
  private double distance(int a, int b) {
    Math.sqrt((0..<3).collect { int i -> double d=position[a][i]-position[b][i]; d*d }.sum() as double)
  }
  private static String key(int a, int b) { "${Math.min(a,b)}-${Math.max(a,b)}" }
}
